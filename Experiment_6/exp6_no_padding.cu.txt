
#include <cstdio>
#include <cstdlib>
#include <cmath>
#include <cuda_runtime.h>

#define TILE 32
#define BLK 16

#define CHECK(call) do { \
    cudaError_t err = (call); \
    if (err != cudaSuccess) { \
        fprintf(stderr, "CUDA Error: %s\n", \
                cudaGetErrorString(err)); \
        exit(1); \
    } \
} while (0)

// KERNEL 1: UNCOALESCED TRANSPOSE

__global__ void transposeNaive(
    const float *in, float *out, int N)
{
    int x = blockIdx.x * TILE + threadIdx.x;
    int y = blockIdx.y * TILE + threadIdx.y;

    if (x < N && y < N)
        out[x * N + y] = in[y * N + x];
}

// KERNEL 2: COALESCED TRANSPOSE

__global__ void transposeCoalesced(
    const float *in, float *out, int N)
{
    __shared__ float tile[TILE][TILE];

    int x = blockIdx.x * TILE + threadIdx.x;
    int y = blockIdx.y * TILE + threadIdx.y;

    if (x < N && y < N)
        tile[threadIdx.y][threadIdx.x] =
            in[y * N + x];

    __syncthreads();

    x = blockIdx.y * TILE + threadIdx.x;
    y = blockIdx.x * TILE + threadIdx.y;

    if (x < N && y < N)
        out[y * N + x] =
            tile[threadIdx.x][threadIdx.y];
}

// KERNEL 3: NAIVE MATRIX MULTIPLICATION

__global__ void matMulNaive(
    const float *A, const float *B,
    float *C, int N)
{
    int row = blockIdx.y * BLK + threadIdx.y;
    int col = blockIdx.x * BLK + threadIdx.x;

    if (row < N && col < N) {
        float sum = 0.0f;

        for (int k = 0; k < N; k++)
            sum += A[row * N + k] *
                   B[k * N + col];

        C[row * N + col] = sum;
    }
}

// CUDA EVENT TIMING

template <bool Coalesced>
float timeTranspose(
    dim3 grid, dim3 block,
    const float *in, float *out,
    int N, int reps)
{
    cudaEvent_t start, stop;

    CHECK(cudaEventCreate(&start));
    CHECK(cudaEventCreate(&stop));

    CHECK(cudaMemset(
        out, 0, (size_t)N * N * sizeof(float)));

    // Warm-up execution

    if (Coalesced)
        transposeCoalesced<<<grid, block>>>(in, out, N);
    else
        transposeNaive<<<grid, block>>>(in, out, N);

    CHECK(cudaGetLastError());
    CHECK(cudaDeviceSynchronize());

    CHECK(cudaEventRecord(start));

    for (int r = 0; r < reps; r++) {
        if (Coalesced)
            transposeCoalesced<<<grid, block>>>(in, out, N);
        else
            transposeNaive<<<grid, block>>>(in, out, N);
    }

    CHECK(cudaGetLastError());
    CHECK(cudaEventRecord(stop));
    CHECK(cudaEventSynchronize(stop));

    float ms;

    CHECK(cudaEventElapsedTime(&ms, start, stop));

    CHECK(cudaEventDestroy(start));
    CHECK(cudaEventDestroy(stop));

    return ms / reps;
}

// CPU VERIFICATION: TRANSPOSE

bool checkTranspose(
    const float *in, const float *out, int N)
{
    for (int i = 0; i < N; i++)
        for (int j = 0; j < N; j++)
            if (out[j * N + i] != in[i * N + j])
                return false;

    return true;
}

// CPU VERIFICATION: MATRIX MULTIPLICATION

bool checkMatMul(
    const float *A, const float *B,
    const float *C, int N)
{
    for (int t = 0; t < 20; t++) {
        int i = rand() % N;
        int j = rand() % N;

        double ref = 0.0;

        for (int k = 0; k < N; k++)
            ref += (double)A[i * N + k] *
                   B[k * N + j];

        if (fabs(ref - C[i * N + j]) >
            1e-3 * fabs(ref) + 1e-3)
            return false;
    }

    return true;
}

// MAIN PROGRAM

int main(int argc, char **argv)
{
    int NT = 4096;
    int NM = 1024;

    if (argc > 1) NT = atoi(argv[1]);
    if (argc > 2) NM = atoi(argv[2]);

    if (NT <= 0 || NM <= 0) {
        fprintf(stderr, "Invalid matrix size\n");
        return 1;
    }

    cudaDeviceProp prop;
    CHECK(cudaGetDeviceProperties(&prop, 0));

    printf("========================================\n");
    printf("EXPERIMENT 6: CUDA MEMORY OPTIMIZATION\n");
    printf("========================================\n");

    printf("GPU: %s\n", prop.name);
    printf("Compute Capability: %d.%d\n",
           prop.major, prop.minor);

    // PART A: MATRIX TRANSPOSE

    size_t bytesT = (size_t)NT * NT * sizeof(float);

    float *h_in = (float *)malloc(bytesT);
    float *h_out = (float *)malloc(bytesT);

    if (!h_in || !h_out) return 1;

    for (size_t i = 0; i < (size_t)NT * NT; i++)
        h_in[i] = (float)(i % 1000);

    float *d_in, *d_out;

    CHECK(cudaMalloc(&d_in, bytesT));
    CHECK(cudaMalloc(&d_out, bytesT));

    CHECK(cudaMemcpy(d_in, h_in, bytesT,
                     cudaMemcpyHostToDevice));

    dim3 blockT(TILE, TILE);

    dim3 gridT(
        (NT + TILE - 1) / TILE,
        (NT + TILE - 1) / TILE);

    float tN = timeTranspose<false>(
        gridT, blockT, d_in, d_out, NT, 20);

    CHECK(cudaMemcpy(h_out, d_out, bytesT,
                     cudaMemcpyDeviceToHost));

    bool okN = checkTranspose(h_in, h_out, NT);

    float tC = timeTranspose<true>(
        gridT, blockT, d_in, d_out, NT, 20);

    CHECK(cudaMemcpy(h_out, d_out, bytesT,
                     cudaMemcpyDeviceToHost));

    bool okC = checkTranspose(h_in, h_out, NT);

    double gbN = 2.0 * bytesT / (tN * 1e6);
    double gbC = 2.0 * bytesT / (tC * 1e6);

    printf("\n=== PART A: MATRIX TRANSPOSE ===\n");
    printf("Matrix size: %d x %d\n", NT, NT);

    printf("Uncoalesced: %.4f ms | %.2f GB/s | %s\n",
           tN, gbN, okN ? "PASS" : "FAIL");

    printf("Coalesced  : %.4f ms | %.2f GB/s | %s\n",
           tC, gbC, okC ? "PASS" : "FAIL");

    printf("Speedup    : %.2fx\n", tN / tC);

    // PART B: NAIVE MATRIX MULTIPLICATION

    size_t bytesM = (size_t)NM * NM * sizeof(float);

    float *hA = (float *)malloc(bytesM);
    float *hB = (float *)malloc(bytesM);
    float *hC = (float *)malloc(bytesM);

    if (!hA || !hB || !hC) return 1;

    for (size_t i = 0; i < (size_t)NM * NM; i++) {
        hA[i] = rand() / (float)RAND_MAX;
        hB[i] = rand() / (float)RAND_MAX;
    }

    float *dA, *dB, *dC;

    CHECK(cudaMalloc(&dA, bytesM));
    CHECK(cudaMalloc(&dB, bytesM));
    CHECK(cudaMalloc(&dC, bytesM));

    CHECK(cudaMemcpy(dA, hA, bytesM,
                     cudaMemcpyHostToDevice));

    CHECK(cudaMemcpy(dB, hB, bytesM,
                     cudaMemcpyHostToDevice));

    dim3 block(BLK, BLK);

    dim3 grid(
        (NM + BLK - 1) / BLK,
        (NM + BLK - 1) / BLK);

    cudaEvent_t start, stop;

    CHECK(cudaEventCreate(&start));
    CHECK(cudaEventCreate(&stop));

    matMulNaive<<<grid, block>>>(dA, dB, dC, NM);

    CHECK(cudaGetLastError());
    CHECK(cudaDeviceSynchronize());

    int reps = 5;

    CHECK(cudaEventRecord(start));

    for (int r = 0; r < reps; r++)
        matMulNaive<<<grid, block>>>(dA, dB, dC, NM);

    CHECK(cudaGetLastError());
    CHECK(cudaEventRecord(stop));
    CHECK(cudaEventSynchronize(stop));

    float ms;

    CHECK(cudaEventElapsedTime(&ms, start, stop));

    float tM = ms / reps;

    CHECK(cudaEventDestroy(start));
    CHECK(cudaEventDestroy(stop));

    CHECK(cudaMemcpy(hC, dC, bytesM,
                     cudaMemcpyDeviceToHost));

    bool okM = checkMatMul(hA, hB, hC, NM);

    double gflops =
        2.0 * NM * NM * (double)NM / (tM * 1e6);

    printf("\n=== PART B: NAIVE MATRIX MULTIPLICATION ===\n");
    printf("Matrix size: %d x %d\n", NM, NM);

    printf("Naive MatMul: %.4f ms | %.2f GFLOPS | %s\n",
           tM, gflops, okM ? "PASS" : "FAIL");

    // FINAL SUMMARY

    printf("\n============== SUMMARY ==============\n");

    printf("Uncoalesced Transpose : %.4f ms\n", tN);
    printf("Coalesced Transpose   : %.4f ms\n", tC);
    printf("Naive MatMul          : %.4f ms\n", tM);
    printf("Transpose Speedup     : %.2fx\n", tN / tC);

    printf("CPU Verification      : %s\n",
           (okN && okC && okM) ? "PASSED" : "FAILED");

    printf("=====================================\n");

    cudaFree(d_in);
    cudaFree(d_out);
    cudaFree(dA);
    cudaFree(dB);
    cudaFree(dC);

    free(h_in);
    free(h_out);
    free(hA);
    free(hB);
    free(hC);

    return (okN && okC && okM) ? 0 : 1;
}
